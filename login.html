import { db } from "./firebase.js"; import { collection, query, where, getDocs, doc, getDoc, updateDoc, addDoc, serverTimestamp } from "firebase/firestore"; /* ========================= GLOBAL STATE ========================= */ let selectedDate = ""; let slotsCache = []; /* ========================= 1) LOAD SLOTS BY DATE ========================= */ window.loadSlots = async function (date) { selectedDate = date; const q = query( collection(db, "slots"), where("date", "==", date) ); const snap = await getDocs(q); slotsCache = []; snap.forEach(d => { slotsCache.push({ id: d.id, ...d.data() }); }); renderSlots(); }; /* ========================= 2) BOOK SLOT (USER FLOW) ========================= */ window.bookSlot = async function (slotId) { const name = document.getElementById("name").value.trim(); const phone = document.getElementById("phone").value.trim(); if (!name || !phone) { alert("من فضلك املأ البيانات"); return; } const ref = doc(db, "slots", slotId); const snap = await getDoc(ref); if (!snap.exists()) { alert("الموعد غير موجود"); return; } const data = snap.data(); /* منع الحجز المكرر */ if (data.booked) { alert("هذا الموعد محجوز بالفعل"); return; } await updateDoc(ref, { booked: true, status: "booked", patient: { name, phone }, updatedAt: serverTimestamp() }); alert("تم الحجز بنجاح ✔️"); loadSlots(selectedDate); }; /* ========================= 3) CREATE REQUEST (NO SLOTS CASE) ========================= */ window.createRequest = async function () { const name = document.getElementById("name").value.trim(); const phone = document.getElementById("phone").value.trim(); if (!name || !phone || !selectedDate) { alert("من فضلك املأ البيانات"); return; } await addDoc(collection(db, "requests"), { name, phone, date: selectedDate, status: "pending", createdAt: serverTimestamp() }); alert("تم إرسال طلبك ✔️ وسيتم تحديد موعد قريبًا"); }; /* ========================= 4) RENDER UI LOGIC ========================= */ function renderSlots() { const container = document.getElementById("slotsContainer"); container.innerHTML = ""; /* لا يوجد slots */ if (slotsCache.length === 0) { container.innerHTML = `

❌ لا يوجد مواعيد متاحة لهذا اليوم

`; return; } /* يوجد slots */ slotsCache .filter(slot => !slot.booked) .sort((a, b) => a.time.localeCompare(b.time)) .forEach(slot => { container.innerHTML += `
🕒 ${slot.time}
`; }); } /* ========================= 5) HELPER (OPTIONAL) ========================= */ window.getSelectedDate = function () { return selectedDate; };
